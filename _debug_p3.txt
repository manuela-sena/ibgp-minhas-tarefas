    </button>

  document.body.appendChild(modal);

  // Carregar buckets
  try {
    await getBuckets();
    const sorted = Object.entries(_buckets).sort((a,b)=>a[1].localeCompare(b[1]));
    window._retroBuckets = sorted; // guardar para o autocomplete
    const inp = document.getElementById('inp-retro-concurso');
    if(inp) inp.placeholder = `Buscar entre ${sorted.length} concursos...`;

    // Inicializar dropdown genérico
    dropdownInit('retro-concurso',
      ()=>sorted.map(([id,nome])=>({label:nome,value:id})),
      (val)=>{
        document.getElementById('retro-btn-criar').disabled = !val;
        if(val) previewRetroativo(val);
        else document.getElementById('retro-preview').innerHTML = '';
      }
    );
  } catch(e){
    document.getElementById('retro-preview').innerHTML = `<div style="color:#c0322f;font-size:12.5px">Erro ao carregar concursos: ${e.message}</div>`;
  }
}
