          if(!v||v==='-') return '-';
          try{const d=new Date(v);if(isNaN(d.getTime())) return v.toString();
            return `${String(d.getDate()).padStart(2,'0')}/${String(d.getMonth()+1).padStart(2,'0')}/${d.getFullYear()}`;}
          catch(e){return v.toString();}
        }
