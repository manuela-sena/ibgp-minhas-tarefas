function calcularTarefasOperacionais(cronograma, temDiscursiva, modo='padrao'){
  const tarefas = [];
  const alertas = [];

  // Extrair datas-âncora do cronograma
  let fimInscricoes=null, dataProva=null, resultPrelObjetiva=null;
  for(const t of cronograma){
    const a=t.atividade.toUpperCase();
    if(a.includes('PERÍODO DE INSCRIÇÕES')&&!fimInscricoes) fimInscricoes=t.data_fim;
    if((a==='PROVA OBJETIVA'||a.startsWith('PROVA OBJETIVA E')||a.startsWith('REALIZAÇÃO DA PROVA OBJETIVA')||a.startsWith('APLICAÇÃO DA PROVA OBJETIVA'))&&!dataProva) dataProva=t.data_fim;
    if(a.includes('RESULTADO PRELIMINAR DA TOTALIZAÇÃO')&&!resultPrelObjetiva) resultPrelObjetiva=t.data_fim;
  }

  if(!dataProva) return {tarefas, alertas};

  // ── Calcular dias úteis disponíveis entre fim inscrições e prova ──
  let duDisponiveis = 0;
  if(fimInscricoes){
    let d = new Date(fimInscricoes);
    while(d < dataProva){ d=addDays(d,1); if(isUtil(d)) duDisponiveis++; }
  }

  // ── Alerta de intervalo curto ──────────────────────────────────────
  if(fimInscricoes && duDisponiveis < 12){
    alertas.push({
      tipo:'risco',
      msg:`⚠️ Intervalo curto: apenas ${duDisponiveis} dias úteis entre fim das inscrições e a prova. Pode não ser possível terminar o fluxo operacional antes da semana da prova.`
    });
  }

  // ── Datas calculadas ───────────────────────────────────────────────
  const quintaAntes = diasUteisAntes(dataProva, 4); // quinta = 4du antes do domingo
  const segunda     = addDays(dataProva, 1);         // segunda pós-prova
  const quarta      = addDays(dataProva, 3);         // quarta pós-prova
  const quinta      = addDays(dataProva, 4);         // quinta pós-prova
  const aposEsc     = addDiasUteis(quarta, 2);       // N1: quarta + 2du

    const A1 = addDiasUteis(fimInscricoes, 1);
    // A2: Receber arquivos → fim insc + 4du
    const A2 = addDiasUteis(fimInscricoes, 4);

    const T1 = addDiasUteis(fimInscricoes, 4);
    // L0: Envio de ofícios às escolas → início das inscrições
    const L0 = new Date(fimInscricoes); // será substituído pela data de início
    // L1: Levantamento escolas → fim insc + 2du
    const L1 = addDiasUteis(fimInscricoes, 2);
    // L2: Alocação → L1 + 2du
    const L2 = addDiasUteis(L1, 2);
    // J1: Impressão provas → A2 + 1du, dura 2du
    const J1ini = addDiasUteis(A2, 1);
    const J1fim = addDiasUteis(J1ini, 2);
    // J2: Impressão papelaria → L2 + 1du, dura 2du (paralelo J1)
    const J2ini = addDiasUteis(L2, 1);
    const J2fim = addDiasUteis(J2ini, 2);
    // J3: Empacotamento → após J1 e J2, dura 2du
    const J3ini = proxUtilApos(J1fim > J2fim ? J1fim : J2fim);
    const J3fim = addDiasUteis(J3ini, 2);
    // J4: Caixas + pastas → J3 + 1du, dura 2du
    const J4ini = addDiasUteis(J3fim, 1);
    const J4fim = addDiasUteis(J4ini, 2);

    // ── Alerta se J4 ultrapassa a quinta anterior à prova ─────────────
    if(J4fim > quintaAntes){
      alertas.push({
        tipo:'atraso',
        msg:`🔴 Fluxo não termina antes da semana da prova. Previsão de conclusão: ${fmtBR(J4fim)} — quinta limite: ${fmtBR(quintaAntes)}.`
      });
    }

    // Buscar data de início das inscrições no cronograma
    let iniInscricoes = null;
    for(const t of cronograma){
      if(t.atividade.toUpperCase().includes('PERÍODO DE INSCRIÇÕES')){
        iniInscricoes = t.data_ini || t.data_fim;
        break;
      }
    }
