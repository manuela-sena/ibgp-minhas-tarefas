class ClfCandidato {
  constructor(inscricao){
    this.inscricao=inscricao; this.nome=''; this.nascimento=null; this.cargo='';
    this.pools={ampla:true}; this.etapas={}; this.eliminado=false; this.motivoEliminacao='';
  }
  get totalGeral(){
    let t=0;
    for(const et of Object.values(this.etapas)){
      if((et.contaNoTotal ?? true) && clfIsNum(et.total)) t+=et.total;
    }
    return t;
  }
}

class ClfMotor {
  constructor(config){ this.config=config; this.candidatos={}; }
  getOrCreate(insc,nome,nasc,cargo){
    let c=this.candidatos[insc];
    if(!c){ c=new ClfCandidato(insc); this.candidatos[insc]=c; }
    if(nome) c.nome=nome;
    if(nasc!=null) c.nascimento=nasc;
    if(cargo) c.cargo=cargo;
    return c;
  }
  carregarFlagsPool(linhas, mapaColunas){
    linhas.forEach(l=>{
      const c=this.getOrCreate(l.inscricao,l.nome,l.nascimento,l.cargo);
      Object.entries(mapaColunas).forEach(([pool,coluna])=>{
        if(!coluna || !(coluna in l)) return;
        c.pools[pool] = clfNorm(l[coluna])==='SIM';
      });
    });
  }
  carregarEtapaPontuacao(nomeEtapa, linhas, campos, corte, contaNoTotal=true){
    linhas.forEach(l=>{
      const c=this.getOrCreate(l.inscricao,l.nome,l.nascimento,l.cargo);
      const valores=l.valores||{};
      const total=l.total;
      let aprovado=true, motivo='';
      if(corte){
        if(!clfIsNum(total)){ aprovado=false; motivo=`${nomeEtapa}: ausente/sem nota numérica`; }
        else{
          if(corte.notaMinimaTotal!=null && total<corte.notaMinimaTotal){ aprovado=false; motivo=`${nomeEtapa}: total ${total} < mínimo ${corte.notaMinimaTotal}`; }
          if(aprovado && corte.naoZerar){
            for(const campo of campos){
              const v=valores[campo];
              if(clfIsNum(v) && v===0){ aprovado=false; motivo=`${nomeEtapa}: zerou em ${campo}`; break; }
            }
          }
        }
      }
      c.etapas[nomeEtapa] = {campos:valores, total, aprovado, contaNoTotal};
      if(!aprovado){ c.eliminado=true; c.motivoEliminacao=c.motivoEliminacao||motivo; }
    });
  }
  carregarEtapaBinaria(nomeEtapa, linhas, valorAprovado, eliminaDoConcurso, gatePool){
    linhas.forEach(l=>{
      const c=this.getOrCreate(l.inscricao,l.nome,l.nascimento,l.cargo);
      const ok = clfNorm(l.resultado)===clfNorm(valorAprovado);
      c.etapas[nomeEtapa] = {resultado:l.resultado, aprovado:ok, contaNoTotal:false};
      if(gatePool) c.pools[gatePool] = ok && !!c.pools[gatePool];
      if(eliminaDoConcurso && !ok){ c.eliminado=true; c.motivoEliminacao=c.motivoEliminacao||`${nomeEtapa}: não apto/deferido`; }
    });
  }
  chaveDesempate(c, dataRef){
    const chave=[];
    (this.config.desempate||[]).forEach(crit=>{
      if(crit.tipo==='idoso_60'){
        const idade=this._idade(c.nascimento,dataRef);
        chave.push(idade!=null && idade>=60 ? 0:1);
      } else if(crit.tipo==='maior_nota'){
        let v=null;
        for(const et of Object.values(c.etapas)){
          if(et.campos && crit.campo in et.campos){ v=et.campos[crit.campo]; break; }
        }
        chave.push(-(clfIsNum(v)?v:-1e9));
      } else if(crit.tipo==='jurado'){
        chave.push(c.pools.jurado ? 0:1);
      } else if(crit.tipo==='idade_maior'){
        chave.push(c.nascimento ? c.nascimento.getTime() : Infinity);
      } else chave.push(0);
    });
    return chave;
  }
  _idade(nasc,dataRef){
    if(!nasc||!dataRef) return null;
    let anos=dataRef.getFullYear()-nasc.getFullYear();
    const antesAniversario=(dataRef.getMonth()<nasc.getMonth())||(dataRef.getMonth()===nasc.getMonth() && dataRef.getDate()<nasc.getDate());
    if(antesAniversario) anos--;
    return anos;
  }
  classificar(dataRef){
    const aptos = Object.values(this.candidatos).filter(c=>!c.eliminado);
    aptos.sort((a,b)=>{
      if(b.totalGeral !== a.totalGeral) return b.totalGeral - a.totalGeral;
      const ka=this.chaveDesempate(a,dataRef), kb=this.chaveDesempate(b,dataRef);
      for(let i=0;i<Math.max(ka.length,kb.length);i++){
        const va=ka[i]??0, vb=kb[i]??0;
        if(va!==vb) return va-vb;
      }
      return 0;
    });
    const poolsConfig = this.config.pools || ['ampla'];
    const porCargo={};
    aptos.forEach(c=>{ (porCargo[c.cargo]=porCargo[c.cargo]||[]).push(c); });
    const ranking={};
    Object.entries(porCargo).forEach(([cargo,lista])=>{
      ranking[cargo]={};
      poolsConfig.forEach(pool=>{ ranking[cargo][pool]=lista.filter(c=>c.pools[pool]); });
    });
    this.ranking=ranking;
    return {ranking, todosAptos:aptos};
  }
}

// ── LEITURA DE PLANILHAS (header sempre linha 2) ─────────────────────
function clfLerColunas(wb, sheet){
  const ws=wb.Sheets[sheet];
  const aoa=XLSX.utils.sheet_to_json(ws,{header:1,range:1,defval:''});
  if(!aoa.length) return [];
  return aoa[0].map(x=>String(x).trim()).filter(Boolean);
}
function clfLerLinhas(wb, sheet){
  const ws=wb.Sheets[sheet];
  const aoa=XLSX.utils.sheet_to_json(ws,{header:1,range:1,defval:'',raw:false});
  if(aoa.length<2) return [];
  const header=aoa[0].map(x=>String(x).trim());
  return aoa.slice(1).filter(r=>r[0]!=='' && r[0]!=null).map(r=>{
    const o={};
    header.forEach((h,i)=>{ o[h]=r[i]; });
    return o;
  });
}
function clfParseData(v){
  if(!v) return null;
  if(v instanceof Date) return v;
  const s=String(v).trim();
  const m = s.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{2,4})/);
  if(m){ const d=new Date(+m[3]<100?2000+ +m[3]:+m[3], +m[2]-1, +m[1]); return isNaN(d.getTime())?null:d; }
  const d2 = new Date(s);
  return isNaN(d2.getTime()) ? null : d2;
}

// ── CONFIG: LISTAR / CARREGAR / SALVAR NO GITHUB ──────────────────────
async function listarConfigsClassificacao(){
  const sel = document.getElementById('clf-config-select');
  try{
    const r = await fetch(`https://api.github.com/repos/${GH_REPO_CLF}/contents/${CLF_CONFIG_DIR}`,
      {headers:{'Authorization':'Bearer '+GH_TOKEN_CLF}});
    if(!r.ok){ sel.innerHTML='<option value="">Nova configuração...</option>'; return; }
    const arquivos = await r.json();
    const nomes = arquivos.filter(a=>a.name.endsWith('.json')).map(a=>a.name.slice(0,-5));
    sel.innerHTML = '<option value="">Nova configuração...</option>' +
      nomes.map(n=>`<option value="${n}">${n}</option>`).join('');
  }catch(e){ /* silencioso */ }
}

async function carregarConfigClassificacao(){
  const nome = document.getElementById('clf-config-select').value;
  if(!nome){ return; }
  try{
    const r = await fetch(`https://api.github.com/repos/${GH_REPO_CLF}/contents/${CLF_CONFIG_DIR}/${encodeURIComponent(nome)}.json`,
      {headers:{'Authorization':'Bearer '+GH_TOKEN_CLF}});
    if(!r.ok){ alert('Configuração não encontrada.'); return; }
    const d = await r.json();
    const config = JSON.parse(decodeURIComponent(escape(atob(d.content.replace(/\n/g,'')))));
    _clfConfig = config;
    document.getElementById('clf-nome').value = config.nome||'';
    ['pcd','pnp','indigena','quilombola'].forEach(p=>{
      document.getElementById('clf-pool-'+p).checked = (config.pools||[]).includes(p);
    });
    renderDesempateList();
    renderEtapasList();
    document.getElementById('clf-status').innerHTML =
      `<div class="alert alert-success" style="background:#e4f3ea;border:1px solid #a8dbb8;border-radius:9px;padding:10px 14px;color:#2e7d52;font-size:13px">✅ Configuração "${nome}" carregada.</div>`;
  }catch(e){ alert('Erro ao carregar: '+e.message); }
}

function coletarConfigAtual(){
  const pools = ['pcd','pnp','indigena','quilombola'].filter(p=>document.getElementById('clf-pool-'+p)?.checked);
  const desempate = [];
  document.querySelectorAll('#clf-desempate-list .clf-desemp-row').forEach(row=>{
    const idx = row.dataset.idx;
    const tipo = document.getElementById('clf-desemp-tipo-'+idx)?.value;
    const campo = document.getElementById('clf-desemp-campo-'+idx)?.value || '';
    desempate.push({tipo, campo});
  });
  const etapas = [];
  document.querySelectorAll('#clf-etapas-list .clf-etapa-card').forEach(card=>{
    const idx = card.dataset.idx;
    const nome = document.getElementById('clf-etapa-nome-'+idx)?.value || '';
    const tipo = document.getElementById('clf-etapa-tipo-'+idx)?.value;
    const et = {nome, tipo};
    if(tipo==='pontuacao'||tipo==='classificatoria'){
      et.campos = (document.getElementById('clf-etapa-campos-'+idx)?.value||'').split(',').map(s=>s.trim()).filter(Boolean);
      if(tipo==='pontuacao'){
        const temCorte = document.getElementById('clf-etapa-temcorte-'+idx)?.checked;
        if(temCorte){
          et.corte = {
            notaMinimaTotal: parseFloat(document.getElementById('clf-etapa-notamin-'+idx)?.value) || null,
            naoZerar: document.getElementById('clf-etapa-naozerar-'+idx)?.checked || false
          };
        }
      }
    } else if(tipo==='binaria'){
      et.valorAprovado = document.getElementById('clf-etapa-valoraprovado-'+idx)?.value || 'DEFERIDA';
      et.eliminaDoConcurso = document.getElementById('clf-etapa-elimina-'+idx)?.checked || false;
      et.gatePool = document.getElementById('clf-etapa-gatepool-'+idx)?.value || '';
    }
    etapas.push(et);
  });
  return {
    nome: document.getElementById('clf-nome')?.value?.trim()||'',
    pools, desempate, etapas
  };
}

async function salvarConfigClassificacao(){
  const config = coletarConfigAtual();
  if(!config.nome){ alert('Dê um nome ao concurso antes de salvar.'); return; }
  try{
    const path = `${CLF_CONFIG_DIR}/${config.nome}.json`;
    const r0 = await fetch(`https://api.github.com/repos/${GH_REPO_CLF}/contents/${path}`,
      {headers:{'Authorization':'Bearer '+GH_TOKEN_CLF}});
    let sha; if(r0.ok){ sha=(await r0.json()).sha; }
    const content = btoa(unescape(encodeURIComponent(JSON.stringify(config,null,2))));
    const payload = {message:`chore: config classificação ${config.nome}`, content};
    if(sha) payload.sha = sha;
    const r = await fetch(`https://api.github.com/repos/${GH_REPO_CLF}/contents/${path}`,
      {method:'PUT', headers:{'Authorization':'Bearer '+GH_TOKEN_CLF,'Content-Type':'application/json'}, body:JSON.stringify(payload)});
    if(!r.ok) throw new Error('Erro ao salvar no GitHub');
    document.getElementById('clf-status').innerHTML =
      `<div style="background:#e4f3ea;border:1px solid #a8dbb8;border-radius:9px;padding:10px 14px;color:#2e7d52;font-size:13px">✅ Configuração salva!</div>`;
    listarConfigsClassificacao();
  }catch(e){ alert('Erro: '+e.message); }
}

    <button onclick="removeEtapaClassificacao(${i})" style="background:#fdeceb;border:none;color:#c0322f;font-size:14px;cursor:pointer;padding:8px 12px;border-radius:8px;font-weight:600">🗑️ Remover</button>
  </div>
  ${subFields}
  ${uploadArea}
</div>`;
  }).join('');
}

async function onEtapaArquivo(idx, input){
  const file = input.files[0]; if(!file) return;
  carregarSheetJSClf(()=>{
    const reader = new FileReader();
    reader.onload = (e)=>{
      const wb = XLSX.read(e.target.result, {type:'binary', cellDates:true});
      _clfEtapaWb[idx] = {wb, sheetNames: wb.SheetNames, sheet:null, colunas:[], map:{}};
      renderEtapasList();
    };
    reader.readAsBinaryString(file);
  });
}

function onEtapaSheetChange(idx){
  const sheet = document.getElementById('clf-etapa-sheet-'+idx)?.value;
  if(!sheet) return;
  const state = _clfEtapaWb[idx];
  state.sheet = sheet;
  state.colunas = clfLerColunas(state.wb, sheet);
  renderEtapasList();
}

function carregarSheetJSClf(cb){
  if(typeof XLSX!=='undefined'){cb();return;}
  const s=document.createElement('script');
  s.src='https://cdnjs.cloudflare.com/ajax/libs/xlsx/0.18.5/xlsx.full.min.js';
  s.onload=cb; document.head.appendChild(s);
}
function carregarExcelJSClf(cb){
  if(typeof ExcelJS!=='undefined'){cb();return;}
  const s=document.createElement('script');
  s.src='https://cdnjs.cloudflare.com/ajax/libs/exceljs/4.3.0/exceljs.min.js';
  s.onload=cb; document.head.appendChild(s);
}

// ── COLETAR MAPEAMENTO ATUAL DE CADA ETAPA (lido do DOM) ──────────────
function coletarMapaEtapa(idx, tipo, campos, poolsAtivos){
  const g = (id) => document.getElementById(id)?.value || '';
  const map = {
    inscricao: g(`clf-etapa-map-insc-${idx}`),
    nome: g(`clf-etapa-map-nome-${idx}`),
    nascimento: g(`clf-etapa-map-nasc-${idx}`),
    cargo: g(`clf-etapa-map-cargo-${idx}`),
  };
  if(tipo==='pontuacao'||tipo==='classificatoria'){
    map.camposMap = {};
    campos.forEach(campo=>{
      const key = campo.replace(/[^a-zA-Z0-9]/g,'_');
      map.camposMap[campo] = g(`clf-etapa-map-campo-${idx}-${key}`);
    });
    map.total = g(`clf-etapa-map-total-${idx}`);
    map.poolCols = {};
    poolsAtivos.forEach(p=>{ map.poolCols[p] = g(`clf-etapa-map-pool-${idx}-${p}`); });
  } else if(tipo==='binaria'){
    map.resultado = g(`clf-etapa-map-resultado-${idx}`);
  }
  return map;
}

// ── GERAR CLASSIFICAÇÃO FINAL ──────────────────────────────────────────
async function gerarClassificacaoFinal(){
  const statusEl = document.getElementById('clf-status');
  const config = coletarConfigAtual();
  if(!config.nome){ alert('Dê um nome ao concurso.'); return; }
  if(!config.etapas.length){ alert('Adicione ao menos uma etapa.'); return; }

  statusEl.innerHTML = '<div class="loading"><div class="spinner"></div>Processando classificação...</div>';

  try{
    const poolsAtivos = ['ampla', ...config.pools];
    const motor = new ClfMotor({pools: poolsAtivos, desempate: config.desempate});

    const resumoEtapas = [];

    for(let i=0;i<config.etapas.length;i++){
      const et = config.etapas[i];
      const wbState = _clfEtapaWb[i];
      if(!wbState?.sheet){ continue; }
      const map = coletarMapaEtapa(i, et.tipo, et.campos||[], poolsAtivos);
      const linhasRaw = clfLerLinhas(wbState.wb, wbState.sheet);

      const linhas = linhasRaw.map(r=>{
        const insc = map.inscricao ? r[map.inscricao] : null;
        if(insc==null || insc==='') return null;
        const item = {
          inscricao: String(insc).trim(),
          nome: map.nome ? r[map.nome] : '',
          nascimento: map.nascimento ? clfParseData(r[map.nascimento]) : null,
          cargo: map.cargo ? String(r[map.cargo]||'').trim() : '',
        };
        if(et.tipo==='pontuacao' || et.tipo==='classificatoria'){
          const valores = {};
          (et.campos||[]).forEach(campo=>{
            const col = map.camposMap?.[campo];
            const v = col ? r[col] : null;
            valores[campo] = (v!=null && v!=='' && !isNaN(parseFloat(v))) ? parseFloat(v) : v;
          });
          item.valores = valores;
          const totalRaw = map.total ? r[map.total] : null;
          item.total = (totalRaw!=null && totalRaw!=='' && !isNaN(parseFloat(totalRaw))) ? parseFloat(totalRaw) : totalRaw;
        } else if(et.tipo==='binaria'){
          item.resultado = map.resultado ? r[map.resultado] : '';
        }
        return item;
      }).filter(Boolean);

      if(et.tipo==='pontuacao' || et.tipo==='classificatoria'){
        motor.carregarEtapaPontuacao(et.nome||`Etapa ${i+1}`, linhas, et.campos||[],
          et.tipo==='pontuacao' ? (et.corte ? {notaMinimaTotal:et.corte.notaMinimaTotal, naoZerar:et.corte.naoZerar} : null) : null,
          true);
        // Flags de pool (convocação por vaga), se colunas foram mapeadas
        const linhasComPool = linhasRaw.map(r=>{
          const insc = map.inscricao ? r[map.inscricao] : null;
          if(insc==null || insc==='') return null;
          const item = {inscricao:String(insc).trim(), nome:map.nome?r[map.nome]:'', cargo:map.cargo?String(r[map.cargo]||'').trim():''};
          poolsAtivos.forEach(p=>{
            const col = map.poolCols?.[p];
            if(col) item[col] = r[col];
          });
          return item;
        }).filter(Boolean);
        const mapaColsPool = {};
        poolsAtivos.forEach(p=>{ if(map.poolCols?.[p]) mapaColsPool[p]=map.poolCols[p]; });
        if(Object.keys(mapaColsPool).length) motor.carregarFlagsPool(linhasComPool, mapaColsPool);
      } else if(et.tipo==='binaria'){
        motor.carregarEtapaBinaria(et.nome||`Etapa ${i+1}`, linhas, et.valorAprovado||'DEFERIDA', et.eliminaDoConcurso||false, et.gatePool||null);
      }
      resumoEtapas.push(`${et.nome||'Etapa '+(i+1)}: ${linhas.length}`);
    }

    const {ranking, todosAptos} = motor.classificar(new Date());

    // ── Gerar planilha de saída ──────────────────────────────────────
    carregarExcelJSClf(async ()=>{
      const wb = new ExcelJS.Workbook();
      const ws = wb.addWorksheet('CLASSIFICAÇÃO');
      const FONTE='Times New Roman';
      const BORDA={top:{style:'thin',color:{argb:'FF000000'}},bottom:{style:'thin',color:{argb:'FF000000'}},left:{style:'thin',color:{argb:'FF000000'}},right:{style:'thin',color:{argb:'FF000000'}}};

      const cols = ['INSCRIÇÃO','CANDIDATO','DATA DE NASCIMENTO','CARGO',
        ...config.pools.map(p=>p.toUpperCase()), 'TOTAL GERAL',
        ...poolsAtivos.map(p=>`CLASS. ${p.toUpperCase()}`)];

      cols.forEach((c,ci)=>{
        const cell = ws.getCell(1,ci+1);
