function loadReajusteConcursos(){
  await getBuckets();
  const sorted = Object.entries(_buckets).sort((a,b)=>a[1].localeCompare(b[1]));
  window._reajBuckets = sorted;
  initDropdownReaj(sorted);
  const inp = document.getElementById('inp-reaj');
  if(inp) inp.placeholder = `Buscar entre ${sorted.length} concursos...`;
}

function initDropdownReaj(buckets){
  dropdownInit('reaj',
    ()=>buckets.map(([id,nome])=>({label:nome,value:id})),
    (val)=>{ loadReajusteTarefas(); }
  );
}
function filtrarReajDropdown(){ dropdownOpen('reaj'); }
function selecionarReajDropdown(id,nome,el){ dropdownSelect('reaj',id,nome); }
