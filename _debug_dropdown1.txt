function dropdownInit(id, getItems, onSelect, opts={}){
  // id: identificador único (ex: 'concurso', 'reaj', 'retro')
  // getItems: fn() → [{label, value}] ou [string]
  // onSelect: fn(value, label) chamado ao selecionar
  _dropdowns[id] = {getItems, onSelect, opts};
}

function dropdownOpen(id){
  const cfg = _dropdowns[id]; if(!cfg) return;
  const inp = document.getElementById('inp-'+id);
  const drop = document.getElementById('drop-'+id);
  if(!inp||!drop) return;
  const termo = inp.value.toLowerCase();
  const items = cfg.getItems();
  const filtrados = termo ? items.filter(i=>(typeof i==='string'?i:i.label).toLowerCase().includes(termo)) : items;
  
  if(!filtrados.length){
    drop.innerHTML='<div class="drop-item" style="color:#8a9ab5;font-style:italic">Nenhum resultado</div>';
    drop.style.display='block'; return;
  }
  const hidden = document.getElementById('sel-'+id);
  const atual = hidden?.value||'';
  drop.innerHTML = (cfg.opts.allOption ? `<div class="drop-item" onclick="dropdownSelect('${id}','','')" style="color:#8a9ab5">Todos</div>` : '') +
    filtrados.slice(0,100).map(i=>{
      const label = typeof i==='string'?i:i.label;
      const val   = typeof i==='string'?i:i.value;
      return `<div class="drop-item ${val===atual?'active':''}" onclick="dropdownSelect('${id}','${val.replace(/'/g,"\'")}','${label.replace(/'/g,"\'")}')"
             title="${label}">${label}</div>`;
    }).join('');
  drop.style.display='block';
}

function dropdownSelect(id, value, label){
  const cfg = _dropdowns[id]; if(!cfg) return;
  const inp    = document.getElementById('inp-'+id);
  const hidden = document.getElementById('sel-'+id);
  const drop   = document.getElementById('drop-'+id);
  if(inp)    inp.value    = label||value;
  if(hidden) hidden.value = value;
  if(drop)   drop.style.display='none';
  drop?.querySelectorAll('.drop-item').forEach(d=>d.classList.remove('active'));
  if(value) drop?.querySelector(`[title="${label||value}"]`)?.classList.add('active');
  cfg.onSelect(value, label);
}

// Fechar todos os dropdowns ao clicar fora
document.addEventListener('click', e=>{
  Object.keys(_dropdowns).forEach(id=>{
    const inp  = document.getElementById('inp-'+id);
    const drop = document.getElementById('drop-'+id);
    if(drop && drop.style.display!=='none' && inp && !inp.closest('[data-dropdown]')?.contains(e.target) && !drop.contains(e.target)){
      drop.style.display='none';
    }
  });
});
