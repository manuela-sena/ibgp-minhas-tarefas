function renderVisibleGroup(){
  const tasks=_filtered;
  const isG=typeof IS_GESTORA!=='undefined'&&IS_GESTORA;
  const showP=isG&&!(document.getElementById('sel-pessoa')?.value);
  const venc=tasks.filter(t=>t.dias!==null&&t.dias<0);
  const hoje=tasks.filter(t=>t.dias===0);
  const sem=tasks.filter(t=>t.dias!==null&&t.dias>=1&&t.dias<=7);
  const fut=tasks.filter(t=>t.dias===null||t.dias>7);
  const groups={vencidas:venc,hoje,['7d']:sem,futuras:fut};
  const cur=groups[_currentTab]||[];
  const el=document.getElementById('task-list');
  if(!el) return;
  if(!cur.length){
    el.innerHTML='<div class="empty-state"><div class="empty-icon">🎉</div><div class="empty-title">Nenhuma tarefa aqui</div><div class="empty-desc">Tudo certo nesta categoria.</div></div>';
    return;
  }
  el.innerHTML=cur.map(t=>renderTaskRow(t,showP)).join('');
}
