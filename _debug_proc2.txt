async function processarAlocacao(){
  const fileCand = document.getElementById('aloc-file-candidatos').files[0];
  const fileCond = document.getElementById('aloc-file-condespecial').files[0];
  const statusEl = document.getElementById('aloc-processamento-status');

  if(!fileCand){
    statusEl.innerHTML = '<div style="color:#c0322f;padding:10px">Selecione a planilha de candidatos (ALOCAÇÃO).</div>';
    return;
  }

  statusEl.innerHTML = '<div class="loading"><div class="spinner"></div>Processando alocação...</div>';

  const todasEnviadas = (window._alocEscolas||[]).filter(e=>e._status?.enviado);
  const enviadas = todasEnviadas.filter(e=>!window._alocSelecionadas || window._alocSelecionadas.has(e.token));
  if(!todasEnviadas.length){
    statusEl.innerHTML = '<div style="color:#c0322f;padding:10px">Nenhuma escola com formulário enviado. Impossível gerar alocação.</div>';
    return;
  }
  if(!enviadas.length){
    statusEl.innerHTML = '<div style="color:#c0322f;padding:10px">Nenhuma escola selecionada. Marque ao menos uma escola na lista acima para gerar a alocação.</div>';
    return;
  }

  carregarSheetJSAloc(()=>{
    carregarExcelJSAloc(async ()=>{
      try{
        // ── Ler planilha de candidatos (ALOCAÇÃO) ──
        const bufCand = await fileCand.arrayBuffer();
        const wbCand = XLSX.read(bufCand, {type:'array', cellDates:true});
        let candidatos = [];
        for(const nome of wbCand.SheetNames){
          const rows = XLSX.utils.sheet_to_json(wbCand.Sheets[nome], {defval:''});
          candidatos = candidatos.concat(rows);
        }
        candidatos = candidatos.map(r=>{
          const n={};
          Object.keys(r).forEach(k=>{ n[k.trim()]=typeof r[k]==='string'?r[k].trim():r[k]; });
          return n;
        }).filter(c=>c['INSCRIÇÃO']);

        if(!candidatos.length){
          statusEl.innerHTML = '<div style="color:#c0322f;padding:10px">Nenhum candidato encontrado na planilha.</div>';
          return;
        }

        // ── DATA PROVA / TURNO sempre vêm da própria planilha de candidatos
        //    (coluna "DATA REALIZAÇÃO PROVA"), nunca do cadastro da escola ──
        candidatos.forEach(c=>{
          const parsed = parseDataRealizacaoProva(c['DATA REALIZAÇÃO PROVA']);
          c['DATA PROVA'] = parsed.data;
          c['TURNO'] = parsed.turno;
        });

        // ── Ler planilha de Condição Especial (aba COND ESPECIAL) — só DEFERIDA ──
        let deferidos = {};
        let totalSolicitantes = 0;
        if(fileCond){
          const bufCond = await fileCond.arrayBuffer();
          const wbCond = XLSX.read(bufCond, {type:'array', cellDates:true});
          const nomeAba = wbCond.SheetNames.find(n=>{
            const u = n.toUpperCase();
            return u.includes('COND') && u.includes('ESPECIAL');
          }) || wbCond.SheetNames[0];
          // a aba tem uma linha de título antes do cabeçalho (range:1 pula essa linha)
          const condRows = XLSX.utils.sheet_to_json(wbCond.Sheets[nomeAba], {defval:'', range:1});
          condRows.forEach(r=>{
            const insc = String(r['INSCRIÇÃO']||'').trim();
            if(!insc) return;
            totalSolicitantes++;
            const resultado = String(r['RESULTADO']||'').trim().toUpperCase();
            if(resultado==='DEFERIDA'){
              deferidos[insc] = {
                atendimento: String(r['ATENDIMENTO ESPECIAL']||'').trim(),
                justificativa: String(r['JUSTIFICATIVA']||'').trim(),
              };
            }
          });
        }

        // ── Mesclar: só quem está DEFERIDA recebe atendimento especial.
        //    Qualquer valor autodeclarado na planilha de candidatos é ignorado. ──
        candidatos.forEach(c=>{
          const insc = String(c['INSCRIÇÃO']||'').trim();
          const def = deferidos[insc];
          c['ATENDIMENTO ESPECIAL'] = def ? def.atendimento : '';
          c['_justificativaEspecial'] = def ? def.justificativa : '';
        });

        const resultado = executarMotorAlocacao(candidatos, enviadas);
        await gerarPlanilhaAlocacao(resultado);

        const avisoCondEspecial = fileCond
          ? ` · 🧩 ${Object.keys(deferidos).length} de ${totalSolicitantes} condições especiais deferidas aplicadas`
          : ' · ⚠️ nenhuma planilha de Condição Especial enviada — ninguém foi marcado com atendimento especial';

        statusEl.innerHTML = `<div style="background:#e4f3ea;border:1px solid #a8dbb8;border-radius:9px;padding:12px 16px;font-size:13px;color:#2e7d52;font-weight:600">
          ✅ Alocação gerada! ${resultado.alocados.length} candidatos alocados
          ${resultado.naoAlocados.length ? ` · ⚠️ ${resultado.naoAlocados.length} não alocados (capacidade insuficiente)` : ''}
          ${avisoCondEspecial}
        </div>`;
      }catch(err){
        statusEl.innerHTML = `<div style="color:#c0322f;padding:10px">Erro: ${err.message}</div>`;
      }
    });
  });
}

function parseDataRealizacaoProva(valor){
  // Aceita objeto Date (quando o Excel guarda como data real) ou texto no formato
  // Date.toString() do JS ("Sun Jul 26 2026 09:15:28 GMT-0300 (Horário Padrão de Brasília)"),
  // que é como esse campo normalmente vem salvo (como texto) na planilha de candidatos.
  if(!valor) return {data:'', turno:''};
  let dt = null;
  if(valor instanceof Date){
    dt = valor;
  } else {
    const s = String(valor).trim();
    const MESES = {Jan:0,Feb:1,Mar:2,Apr:3,May:4,Jun:5,Jul:6,Aug:7,Sep:8,Oct:9,Nov:10,Dec:11};
    const m = s.match(/^\w{3}\s+(\w{3})\s+(\d{1,2})\s+(\d{4})\s+(\d{2}):(\d{2}):(\d{2})/);
    if(m && MESES[m[1]]!==undefined){
      dt = new Date(Number(m[3]), MESES[m[1]], Number(m[2]), Number(m[4]), Number(m[5]), Number(m[6]));
    } else {
      const tentativa = new Date(s);
      if(!isNaN(tentativa.getTime())) dt = tentativa;
    }
  }
  if(!dt || isNaN(dt.getTime())) return {data:'', turno:''};
  const dd = String(dt.getDate()).padStart(2,'0');
  const mm = String(dt.getMonth()+1).padStart(2,'0');
  const yyyy = dt.getFullYear();
  const turno = dt.getHours() < 12 ? 'MANHÃ' : 'TARDE';
  return {data: `${dd}/${mm}/${yyyy}`, turno};
}

function carregarSheetJSAloc(cb){
  if(typeof XLSX!=='undefined'){cb();return;}
  const s=document.createElement('script');
  s.src='https://cdnjs.cloudflare.com/ajax/libs/xlsx/0.18.5/xlsx.full.min.js';
  s.onload=cb; document.head.appendChild(s);
}

function salaEhTerrea(andarRaw){
  let andar = String(andarRaw==null?'':andarRaw).toUpperCase();
  andar = andar.normalize('NFD').replace(/[̀-ͯ]/g,'');
  andar = andar.replace(/°/g,'º').trim();
  if(andar==='0') return true;
  if(andar.includes('TERREO')) return true;
  if(andar.includes('UNICO')) return true; // prédio de andar único (ex: escola de 1 pavimento só)
  if(/^1\s*[ºO]?\s*ANDAR$/.test(andar)) return true;
  return false;
}
