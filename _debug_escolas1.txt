      const nomeEscola   = (dados.concurso?.escola||dados.escola||'').toString();
      const matchConcurso = !concursoNome || 
        nomeConcurso.toLowerCase().trim().includes(concursoNome.toLowerCase().trim().substring(0,15)) ||
        dados.token?.startsWith('DEMO');
      if(matchConcurso){
        // Buscar status
        const statusFile = statusArq.find(s=>s.name==='status_'+dados.token+'.json');
        let status = {acessos:0, enviado:false};
        if(statusFile){
          const rs = await fetch(statusFile.download_url);
          status = await rs.json();
        }
        // Garantir que enviado está correto — checar tanto no status quanto no token
        if(dados.enviadoEm) status.enviado = true;
        // Normalizar para boolean
        status.enviado = status.enviado === true || status.enviado === 'true';
        escolas.push({...dados, _status: status});
      }
    }
    
    renderEscolas(escolas, concursoNome);
  }catch(e){
    el.innerHTML=`<div class="card card-pad" style="color:#c0322f">Erro ao carregar: ${e.message}</div>`;
