function renderEtapasList(){
  const etapas = _clfConfig.etapas || [];
  const el = document.getElementById('clf-etapas-list');
  if(!etapas.length){
    el.innerHTML = '<div style="font-size:12.5px;color:#8a9ab5;padding:8px 0">Nenhuma etapa configurada.</div>';
    return;
  }
  const poolsAtivos = ['ampla',...['pcd','pnp','indigena','quilombola'].filter(p=>document.getElementById('clf-pool-'+p)?.checked)];

  el.innerHTML = etapas.map((et,i)=>{
    let subFields = '';
    if(et.tipo==='pontuacao' || et.tipo==='classificatoria'){
      subFields = `
      <div style="margin-top:10px">
        <label class="form-label">Campos de nota (separados por vírgula)</label>
        <input type="text" id="clf-etapa-campos-${i}" class="form-input" style="height:34px" placeholder="Ex: PORTUGUÊS,MATEMÁTICA,CONHECIMENTOS ESPECÍFICOS" value="${(et.campos||[]).join(',')}" oninput="syncClfEtapaCampos(${i}, this.value)">
      </div>`;
      if(et.tipo==='pontuacao'){
        subFields += `
      <div style="display:flex;gap:16px;align-items:center;margin-top:10px;flex-wrap:wrap">
        <label style="display:flex;align-items:center;gap:6px;font-size:13px;cursor:pointer">
          <input type="checkbox" id="clf-etapa-temcorte-${i}" ${et.corte?'checked':''} onchange="renderEtapasList()"> Tem nota mínima?
        </label>
        ${et.corte ? `
        <div><label class="form-label">Nota mínima total</label><input type="number" id="clf-etapa-notamin-${i}" class="form-input" style="height:32px;width:110px" value="${et.corte.notaMinimaTotal??''}"></div>
        <label style="display:flex;align-items:center;gap:6px;font-size:13px;cursor:pointer">
          <input type="checkbox" id="clf-etapa-naozerar-${i}" ${et.corte.naoZerar?'checked':''}> Não pode zerar nenhum campo
        </label>` : ''}
      </div>`;
      }
    } else if(et.tipo==='binaria'){
      subFields = `
      <div style="display:grid;grid-template-columns:1fr 1fr 1fr;gap:10px;margin-top:10px">
        <div><label class="form-label">Valor aprovado</label><input type="text" id="clf-etapa-valoraprovado-${i}" class="form-input" style="height:34px" value="${et.valorAprovado||'DEFERIDA'}"></div>
        <div style="display:flex;align-items:flex-end">
          <label style="display:flex;align-items:center;gap:6px;font-size:13px;cursor:pointer;padding-bottom:9px">
            <input type="checkbox" id="clf-etapa-elimina-${i}" ${et.eliminaDoConcurso?'checked':''}> Reprovar elimina do concurso?
          </label>
        </div>
        <div><label class="form-label">Gateia qual pool? (opcional)</label>
          <select id="clf-etapa-gatepool-${i}" class="filter-select" style="height:34px">
            <option value="">Nenhum</option>
            ${poolsAtivos.filter(p=>p!=='ampla').map(p=>`<option value="${p}" ${et.gatePool===p?'selected':''}>${p.toUpperCase()}</option>`).join('')}
          </select>
        </div>
      </div>`;
    }

    const wbState = _clfEtapaWb[i];
    let uploadArea = `
      <div style="margin-top:12px;padding-top:12px;border-top:1px dashed #e2e7ef">
        <label class="form-label">Planilha desta etapa</label>
        <input type="file" accept=".xlsx" onchange="onEtapaArquivo(${i},this)" style="font-size:12.5px">
      </div>`;
    if(wbState?.sheetNames?.length){
      uploadArea += `
      <div style="margin-top:10px">
        <label class="form-label">Abas utilizadas (todas as marcadas serão combinadas nesta etapa)</label>
