        resp:'Natália', ini:proxUtilApos(quarta_pos), fim:addDiasUteis(quarta_pos, 2)},
    );
  }

  // ── Fluxo Discursiva ──────────────────────────────────────────────
  if(temDiscursiva && resultPrelObjetiva){
    const D1=proxUtilApos(resultPrelObjetiva);
    const D2=proxUtilApos(D1);
    tarefas.push(
      {titulo:'SEPARAÇÃO DAS PROVAS DISCURSIVAS PARA CORREÇÃO', resp:'Natália', ini:D1, fim:D1},
      {titulo:'ENVIO DAS PROVAS DISCURSIVAS PARA A BANCA',      resp:'Natália', ini:proxUtilApos(D1), fim:D2},
    );
  }

  // ── Despacho do material → após classificação preliminar ──────────
  if(resultPrelObjetiva){
    const despacho = proxUtilApos(resultPrelObjetiva);
    tarefas.push(
      {titulo:'DESPACHO DO MATERIAL PARA O MUNICÍPIO', resp:'Jordan', ini:despacho, fim:despacho},
    );
  }

  return {tarefas, alertas};
}
