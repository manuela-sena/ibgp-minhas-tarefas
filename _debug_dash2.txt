async function apiAll(url) {
  let res=[], next=url;
  while(next){ const d=await api(next); res=res.concat(d.value||[]); next=d['@odata.nextLink']||null; }
  return res;
}
async function apiPatch(url, body) {
  const r1=await fetch(url,{headers:{'Authorization':'Bearer '+TOKEN}});
  const etag=r1.headers.get('ETag')||'*';
  return fetch(url,{method:'PATCH',headers:{'Authorization':'Bearer '+TOKEN,'Content-Type':'application/json','If-Match':etag},body:JSON.stringify(body)});
}

// ── PLANNER CACHE ───────────────────────────────────────────────────
let _planoId=null, _buckets={}, _bucketsLoaded=false;
async function getPlanoId() {
  if(_planoId) return _planoId;
  const groups = await apiAll('https://graph.microsoft.com/v1.0/me/memberOf');
  for(const g of groups){
    try{
      const plans=await api(`https://graph.microsoft.com/v1.0/groups/${g.id}/planner/plans`);
      for(const p of(plans.value||[])){ if(p.title.toUpperCase().includes('PLANNER IBGP')){ _planoId=p.id; return p.id; } }
    }catch(e){}
  }
  return null;
}
async function getBuckets() {
  const pid=await getPlanoId(); if(!pid) return {};
  const bd=await api(`https://graph.microsoft.com/v1.0/planner/plans/${pid}/buckets`);
  (bd.value||[]).forEach(b=>_buckets[b.id]=b.name);
  return _buckets;
}

function chipP(nome){
  const cls={Lorena:'chip-Lorena',Laryssa:'chip-Laryssa','Natália':'chip-Natália',Manuela:'chip-Manuela'};
  return `<span class="chip-p ${cls[nome]||'chip-default'}"><svg width="11" height="11" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><circle cx="5.5" cy="4" r="2.5"/><path d="M1 11a4.5 4.5 0 0 1 9 0"/></svg>${nome}</span>`;
}
function chipD(dias,data){
  if(dias===null) return `<span class="chip-d chip-nodate">Sem data</span>`;
  if(dias<0) return `<span class="chip-d chip-venc">Venceu há ${Math.abs(dias)}d · ${data}</span>`;
  if(dias===0) return `<span class="chip-d chip-hoje">⚡ Hoje</span>`;
  if(dias<=7) return `<span class="chip-d chip-7d">${data} · ${dias}d</span>`;
  return `<span class="chip-d chip-ok">${data} · ${dias}d</span>`;
}
