function addSala(){
  salaCount++;
  const tbody = document.getElementById('salas-tbody');
  const tr = document.createElement('tr');
  tr.id = 'sala-row-'+salaCount;
  tr.innerHTML = `
<td style="text-align:center;color:var(--muted);font-size:12px;font-weight:700">${tbody.children.length+1}</td>
<td><input type="text" data-f="bloco" placeholder="Ex: A, B" style="min-width:70px"></td>
<td><input type="text" data-f="andar" placeholder="1º Andar" style="min-width:90px"></td>
<td><input type="text" data-f="sala" placeholder="Ex: 101" style="min-width:70px"></td>
<td><input type="number" data-f="cap" min="1" max="100" placeholder="0" oninput="atualizarCapTotal()"></td>
<td>
  <select data-f="turno">
    <option value="AMBOS">Manhã e Tarde</option>
    <option value="MANHA">Somente Manhã</option>
    <option value="TARDE">Somente Tarde</option>
  </select>
</td>
<td style="text-align:center"><input type="checkbox" data-f="arcond" style="width:16px;height:16px;accent-color:var(--azul-med)"></td>
<td style="text-align:center"><input type="checkbox" data-f="vent" style="width:16px;height:16px;accent-color:var(--azul-med)"></td>
<td><button class="btn-del" onclick="delSala('sala-row-${salaCount}')" title="Remover">×</button></td>`;
  tbody.appendChild(tr);
  renumSalas();
}

function delSala(id){
  document.getElementById(id)?.remove();
  renumSalas();
  atualizarCapTotal();
}

function renumSalas(){
  document.querySelectorAll('#salas-tbody tr').forEach((r,i)=>{
    r.querySelector('td:first-child').textContent = i+1;
  });
}

function atualizarCapTotal(){
  let total = 0;
  document.querySelectorAll('[data-f="cap"]').forEach(i=>{ total += parseInt(i.value)||0; });
  document.getElementById('total-cap-tag').textContent = total + ' candidatos no total';
}

function buildRevisao(){
  const salas = getSalas();
  const html = `
<div class="card" style="margin-bottom:16px">
  <div class="card-header">
    <div class="card-title">📋 Dados da Instituição</div>
    <button class="btn btn-secondary" style="margin-left:auto;padding:6px 12px;font-size:12px" onclick="goStep(1)">Editar</button>
  </div>
  <div class="card-body">
    <div class="form-grid">
      ${rv('Nome', v('nome'))} ${rv('CNPJ', v('cnpj'))}
      ${rv('Endereço', [v('endereco'),v('numero'),v('bairro'),v('cidade'),v('uf')].filter(Boolean).join(', '))}
      ${rv('Responsável', v('resp-nome'))} ${rv('Telefone', v('resp-tel'))}
      ${rv('E-mail', v('resp-email'))}
    </div>
  </div>
</div>
<div class="card" style="margin-bottom:16px">
  <div class="card-header">
    <div class="card-title">🏫 Salas cadastradas — ${salas.length} sala(s)</div>
    <button class="btn btn-secondary" style="margin-left:auto;padding:6px 12px;font-size:12px" onclick="goStep(3)">Editar</button>
  </div>
  <div class="card-body" style="padding:0">
    <div class="salas-table-wrap">
      <table class="salas">
        <thead><tr><th>#</th><th>Bloco</th><th>Andar</th><th>Sala</th><th>Capacidade</th><th>Turno</th><th>Ar</th><th>Vent.</th></tr></thead>
        <tbody>${salas.map((s,i)=>`<tr>
          <td style="text-align:center">${i+1}</td>
          <td>${s.bloco||'—'}</td><td>${s.andar||'—'}</td><td><strong>${s.sala}</strong></td>
          <td style="text-align:center"><strong>${s.capacidade}</strong></td>
          <td>${s.turno==='AMBOS'?'Manhã e Tarde':s.turno==='MANHA'?'Manhã':'Tarde'}</td>
          <td style="text-align:center">${s.arcond?'✅':'—'}</td>
          <td style="text-align:center">${s.ventilador?'✅':'—'}</td>
        </tr>`).join('')}</tbody>
      </table>
    </div>
  </div>
</div>`;
  document.getElementById('revisao-content').innerHTML = html;
}

function rv(label, val){
  return `<div class="form-group"><label style="font-weight:600">${label}</label><div style="font-size:13px;padding:8px 12px;background:var(--cinza);border-radius:7px">${val||'—'}</div></div>`;
}

function getSalas(){
  const rows = document.querySelectorAll('#salas-tbody tr');
  return Array.from(rows).map(r=>({
    bloco:   r.querySelector('[data-f="bloco"]').value.trim(),
    andar:   r.querySelector('[data-f="andar"]').value.trim(),
    sala:    r.querySelector('[data-f="sala"]').value.trim(),
    capacidade: parseInt(r.querySelector('[data-f="cap"]').value)||0,
    turno:   r.querySelector('[data-f="turno"]').value,
    arcond:  r.querySelector('[data-f="arcond"]').checked,
    ventilador: r.querySelector('[data-f="vent"]').checked,
  }));
}
