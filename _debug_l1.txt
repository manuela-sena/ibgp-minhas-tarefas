function onEtapaArquivo(idx, input){
  const file = input.files[0]; if(!file) return;
  carregarSheetJSClf(()=>{
    const reader = new FileReader();
    reader.onload = (e)=>{
      const wb = XLSX.read(e.target.result, {type:'binary', cellDates:true});
      const sheetNames = wb.SheetNames;
      _clfEtapaWb[idx] = {wb, sheetNames, sheets: [...sheetNames], colunas: clfLerColunas(wb, sheetNames[0]), map:{}};
      renderEtapasList();
    };
    reader.readAsBinaryString(file);
  });
}

function onEtapaSheetToggle(idx, sheet, checked){
  const state = _clfEtapaWb[idx];
  if(!state) return;
  state.sheets = state.sheets || [];
  if(checked){ if(!state.sheets.includes(sheet)) state.sheets.push(sheet); }
  else { state.sheets = state.sheets.filter(s=>s!==sheet); }
  state.sheets = state.sheetNames.filter(s=>state.sheets.includes(s));
  state.colunas = state.sheets.length ? clfLerColunas(state.wb, state.sheets[0]) : [];
  renderEtapasList();
}
