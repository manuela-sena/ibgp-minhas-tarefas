}`;
          tCell.font={bold:true,color:{argb:'FFFFFFFF'},name:FONTE,size:11};
          tCell.fill={type:'pattern',pattern:'solid',fgColor:{argb:AZUL_ESC}};
          tCell.alignment={horizontal:'center',vertical:'middle',wrapText:true};
          tCell.border=BORDA;
          ws.getRow(1).height=60;

          // Linha 2: cabeçalho
          colsSaida.forEach((col,ci)=>{
            const c=ws.getCell(2,ci+1);
            c.value=col;
            c.font={bold:true,color:{argb:'FFFFFFFF'},name:FONTE,size:9};
            c.fill={type:'pattern',pattern:'solid',fgColor:{argb:AZUL_MED}};
            c.alignment={horizontal:'center',vertical:'middle',wrapText:true};
            c.border=BORDA;
            ws.getColumn(ci+1).width=largDefault[col]||12;
          });
          ws.getRow(2).height=25;

          // Dados
          let cargoCorAtual='',corIdx=0;
          saida.forEach((row,ri)=>{
            if(row._cargo!==cargoCorAtual){cargoCorAtual=row._cargo;corIdx=(corIdx+1)%2;}
            const cor=corIdx===0?COR_A:COR_B;
            const isFirst=ri>0&&saida[ri-1]._cargo!==row._cargo;
            colsSaida.forEach((col,ci)=>{
              const c=ws.getCell(ri+3,ci+1);
              let val=col==='CLASS.'?row['CLASS.']:(row._dados[col]??'');
              if(val===undefined||val===null||String(val)==='nan') val='';
              if(col.includes('NASC')&&val){
                try{const d=new Date(val);if(!isNaN(d.getTime())) val=`${String(d.getDate()).padStart(2,'0')}/${String(d.getMonth()+1).padStart(2,'0')}/${d.getFullYear()}`;}catch(e){}
              }
              c.value=col==='CLASS.'?(parseInt(val)||''):String(val);
              c.font={name:FONTE,size:9};
              c.fill={type:'pattern',pattern:'solid',fgColor:{argb:cor}};
              c.alignment={vertical:'middle',horizontal:col==='CLASS.'||col==='CÓDIGO'?'center':'left'};
              c.border=isFirst
                ? {...BORDA,top:{style:'medium',color:{argb:'FF1F3A6B'}}}
                : BORDA;
            });
            ws.getRow(ri+3).height=16;
          });

          ws.views=[{state:'frozen',xSplit:0,ySplit:2}];
          resumo.push(`${aba}: ${saida.length}`);
        }

        // Download
        const bufH=await wb.xlsx.writeBuffer();
        const blobH=new Blob([bufH],{type:'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'});
        const aH=document.createElement('a');
        aH.href=URL.createObjectURL(blobH);
        aH.download='HOMOLOGACAO.xlsx';
        aH.click();

        res.innerHTML=`<div style="background:#e4f3ea;border:1px solid #a8dbb8;border-radius:9px;padding:12px 16px;font-size:13px;color:#2e7d52;font-weight:600">
          ✅ Homologação gerada!<br>
          <span style="font-weight:400;font-size:12px">${resumo.join(' · ')}</span>
        </div>`;
      }catch(e){
        res.innerHTML=`<div style="color:#c0322f;padding:12px">Erro: ${e.message}</div>`;
      }
    });
  }catch(e){
    res.innerHTML=`<div style="color:#c0322f;padding:12px">Erro: ${e.message}</div>`;
  }
